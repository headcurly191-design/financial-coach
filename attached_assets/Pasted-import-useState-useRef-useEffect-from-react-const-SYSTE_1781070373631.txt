import { useState, useRef, useEffect } from "react";

const SYSTEM_PROMPT = `You are ATLAS — an elite personal financial coach with deep expertise in:
- Personal budgeting, saving strategies, and debt management
- Stock market analysis, sector trends, and investment fundamentals
- Crypto, ETFs, index funds, and portfolio diversification
- Reading and interpreting live market data, earnings reports, and economic indicators
- Goal-based financial planning (emergency funds, retirement, wealth building)

You have access to real-time web search. When the user asks about:
- Stock prices, market trends, indices (S&P 500, NASDAQ, etc.) → always search for current data
- Crypto prices or trends → search for current data
- Economic news, Fed decisions, inflation data → search for current data
- Specific company financials or news → search for current data

Your communication style:
- Direct, confident, data-driven — like a Bloomberg analyst meets personal advisor
- Lead with the key insight, then support it with data
- Use short paragraphs. Never ramble.
- When referencing market data, cite the numbers
- Flag risks clearly without being alarmist
- Never give advice framed as "you must do this" — frame as analysis and options
- Keep responses concise for mobile reading

When you search and get results, synthesize them into clear actionable insight. Don't just dump raw data.`;

const WEB_SEARCH_TOOL = {
  type: "web_search_20250305",
  name: "web_search",
};

function TypingIndicator() {
  return (
    <div style={{ display: "flex", alignItems: "center", gap: 6, padding: "12px 16px" }}>
      {[0, 1, 2].map((i) => (
        <div
          key={i}
          style={{
            width: 7,
            height: 7,
            borderRadius: "50%",
            background: "#c8a96e",
            animation: `pulse 1.2s ease-in-out ${i * 0.2}s infinite`,
          }}
        />
      ))}
    </div>
  );
}

function Message({ msg }) {
  const isUser = msg.role === "user";
  return (
    <div
      style={{
        display: "flex",
        justifyContent: isUser ? "flex-end" : "flex-start",
        marginBottom: 16,
        padding: "0 16px",
      }}
    >
      {!isUser && (
        <div
          style={{
            width: 32,
            height: 32,
            borderRadius: "50%",
            background: "linear-gradient(135deg, #c8a96e, #8b6914)",
            display: "flex",
            alignItems: "center",
            justifyContent: "center",
            fontSize: 13,
            fontWeight: 700,
            color: "#0a0a0f",
            marginRight: 10,
            flexShrink: 0,
            marginTop: 2,
            fontFamily: "'Space Mono', monospace",
          }}
        >
          A
        </div>
      )}
      <div
        style={{
          maxWidth: "78%",
          padding: "12px 16px",
          borderRadius: isUser ? "18px 18px 4px 18px" : "4px 18px 18px 18px",
          background: isUser
            ? "linear-gradient(135deg, #c8a96e22, #c8a96e33)"
            : "#13131a",
          border: isUser ? "1px solid #c8a96e55" : "1px solid #1e1e2a",
          color: "#e8e4d8",
          fontSize: 14,
          lineHeight: 1.65,
          fontFamily: "'Inter', sans-serif",
          whiteSpace: "pre-wrap",
          wordBreak: "break-word",
        }}
      >
        {msg.content}
        {msg.searching && (
          <div
            style={{
              marginTop: 8,
              fontSize: 11,
              color: "#c8a96e",
              fontFamily: "'Space Mono', monospace",
              display: "flex",
              alignItems: "center",
              gap: 6,
            }}
          >
            <span style={{ animation: "spin 1s linear infinite", display: "inline-block" }}>⟳</span>
            SEARCHING MARKETS...
          </div>
        )}
      </div>
    </div>
  );
}

function QuickPrompt({ text, onClick }) {
  return (
    <button
      onClick={() => onClick(text)}
      style={{
        background: "#0d0d14",
        border: "1px solid #1e1e2a",
        borderRadius: 20,
        padding: "8px 14px",
        color: "#9090a8",
        fontSize: 12,
        fontFamily: "'Inter', sans-serif",
        cursor: "pointer",
        whiteSpace: "nowrap",
        transition: "all 0.2s",
        flexShrink: 0,
      }}
      onMouseEnter={(e) => {
        e.target.style.borderColor = "#c8a96e55";
        e.target.style.color = "#c8a96e";
      }}
      onMouseLeave={(e) => {
        e.target.style.borderColor = "#1e1e2a";
        e.target.style.color = "#9090a8";
      }}
    >
      {text}
    </button>
  );
}

export default function FinancialCoach() {
  const [messages, setMessages] = useState([
    {
      role: "assistant",
      content:
        "ATLAS online. I'm your financial intelligence system — I can analyze live market conditions, build you a budget strategy, break down investments, or help you understand what's moving the markets right now.\n\nWhat do you want to work on?",
    },
  ]);
  const [input, setInput] = useState("");
  const [loading, setLoading] = useState(false);
  const messagesEndRef = useRef(null);
  const inputRef = useRef(null);

  const quickPrompts = [
    "How's the S&P 500 today?",
    "Best ETFs for beginners?",
    "Build me a budget plan",
    "Is now a good time to invest?",
    "Top trending stocks",
    "How to start an emergency fund?",
  ];

  useEffect(() => {
    messagesEndRef.current?.scrollIntoView({ behavior: "smooth" });
  }, [messages]);

  async function sendMessage(text) {
    const userText = text || input.trim();
    if (!userText || loading) return;

    setInput("");
    const newMessages = [...messages, { role: "user", content: userText }];
    setMessages(newMessages);
    setLoading(true);

    // Add typing indicator
    setMessages((prev) => [
      ...prev,
      { role: "assistant", content: "", typing: true },
    ]);

    try {
      const apiMessages = newMessages.map((m) => ({
        role: m.role,
        content: m.content,
      }));

      const response = await fetch("https://api.anthropic.com/v1/messages", {
        method: "POST",
        headers: { "Content-Type": "application/json" },
        body: JSON.stringify({
          model: "claude-sonnet-4-20250514",
          max_tokens: 1000,
          system: SYSTEM_PROMPT,
          tools: [WEB_SEARCH_TOOL],
          messages: apiMessages,
        }),
      });

      const data = await response.json();

      // Extract the final text response
      let finalText = "";
      let didSearch = false;

      if (data.content) {
        // Check if web search was used
        for (const block of data.content) {
          if (block.type === "tool_use" && block.name === "web_search") {
            didSearch = true;
          }
        }

        // If search was triggered, show "searching" then do follow-up
        if (didSearch) {
          setMessages((prev) => {
            const updated = [...prev];
            updated[updated.length - 1] = {
              role: "assistant",
              content: "Pulling live market data...",
              searching: true,
            };
            return updated;
          });

          // Build full conversation for follow-up with tool results
          const assistantMsg = { role: "assistant", content: data.content };
          const toolResults = data.content
            .filter((b) => b.type === "tool_result" || b.type === "tool_use")
            .map((b) => b);

          // Re-request with tool results included
          const followUp = await fetch("https://api.anthropic.com/v1/messages", {
            method: "POST",
            headers: { "Content-Type": "application/json" },
            body: JSON.stringify({
              model: "claude-sonnet-4-20250514",
              max_tokens: 1000,
              system: SYSTEM_PROMPT,
              tools: [WEB_SEARCH_TOOL],
              messages: [
                ...apiMessages,
                { role: "assistant", content: data.content },
              ],
            }),
          });

          const followData = await followUp.json();
          if (followData.content) {
            for (const block of followData.content) {
              if (block.type === "text") {
                finalText += block.text;
              }
            }
          }
        } else {
          for (const block of data.content) {
            if (block.type === "text") {
              finalText += block.text;
            }
          }
        }
      }

      setMessages((prev) => {
        const updated = [...prev];
        updated[updated.length - 1] = {
          role: "assistant",
          content: finalText || "I couldn't retrieve that information. Try rephrasing.",
        };
        return updated;
      });
    } catch (err) {
      setMessages((prev) => {
        const updated = [...prev];
        updated[updated.length - 1] = {
          role: "assistant",
          content: "Connection issue. Check your API key and try again.",
        };
        return updated;
      });
    }

    setLoading(false);
  }

  function handleKeyDown(e) {
    if (e.key === "Enter" && !e.shiftKey) {
      e.preventDefault();
      sendMessage();
    }
  }

  return (
    <div
      style={{
        display: "flex",
        flexDirection: "column",
        height: "100vh",
        background: "#07070e",
        fontFamily: "'Inter', sans-serif",
        position: "relative",
        overflow: "hidden",
      }}
    >
      <style>{`
        @import url('https://fonts.googleapis.com/css2?family=Space+Mono:wght@400;700&family=Inter:wght@400;500;600&display=swap');
        
        * { box-sizing: border-box; margin: 0; padding: 0; }
        
        @keyframes pulse {
          0%, 100% { opacity: 0.3; transform: scale(0.8); }
          50% { opacity: 1; transform: scale(1.1); }
        }
        
        @keyframes spin {
          from { transform: rotate(0deg); }
          to { transform: rotate(360deg); }
        }
        
        @keyframes fadeIn {
          from { opacity: 0; transform: translateY(8px); }
          to { opacity: 1; transform: translateY(0); }
        }

        ::-webkit-scrollbar { width: 4px; }
        ::-webkit-scrollbar-track { background: transparent; }
        ::-webkit-scrollbar-thumb { background: #1e1e2a; border-radius: 4px; }

        textarea::placeholder { color: #444455; }
        textarea { resize: none; }
      `}</style>

      {/* Ambient glow */}
      <div
        style={{
          position: "absolute",
          top: -100,
          right: -100,
          width: 400,
          height: 400,
          borderRadius: "50%",
          background: "radial-gradient(circle, #c8a96e08 0%, transparent 70%)",
          pointerEvents: "none",
        }}
      />

      {/* Header */}
      <div
        style={{
          padding: "16px 20px",
          borderBottom: "1px solid #1a1a24",
          display: "flex",
          alignItems: "center",
          gap: 12,
          background: "#07070e",
          zIndex: 10,
        }}
      >
        <div
          style={{
            width: 40,
            height: 40,
            borderRadius: "50%",
            background: "linear-gradient(135deg, #c8a96e, #7a5c10)",
            display: "flex",
            alignItems: "center",
            justifyContent: "center",
            fontSize: 16,
            fontWeight: 700,
            color: "#0a0a0f",
            fontFamily: "'Space Mono', monospace",
            boxShadow: "0 0 20px #c8a96e30",
          }}
        >
          A
        </div>
        <div>
          <div
            style={{
              fontSize: 15,
              fontWeight: 600,
              color: "#e8e4d8",
              fontFamily: "'Space Mono', monospace",
              letterSpacing: "0.05em",
            }}
          >
            ATLAS
          </div>
          <div
            style={{
              fontSize: 11,
              color: "#4ade8055",
              fontFamily: "'Space Mono', monospace",
              display: "flex",
              alignItems: "center",
              gap: 5,
            }}
          >
            <span
              style={{
                width: 6,
                height: 6,
                borderRadius: "50%",
                background: "#4ade80",
                display: "inline-block",
                boxShadow: "0 0 6px #4ade80",
              }}
            />
            <span style={{ color: "#4ade80" }}>LIVE MARKET ACCESS</span>
          </div>
        </div>
        <div style={{ marginLeft: "auto", textAlign: "right" }}>
          <div
            style={{
              fontSize: 10,
              color: "#333344",
              fontFamily: "'Space Mono', monospace",
              letterSpacing: "0.1em",
            }}
          >
            FINANCIAL INTELLIGENCE
          </div>
          <div
            style={{
              fontSize: 10,
              color: "#333344",
              fontFamily: "'Space Mono', monospace",
            }}
          >
            v2.1
          </div>
        </div>
      </div>

      {/* Messages */}
      <div
        style={{
          flex: 1,
          overflowY: "auto",
          paddingTop: 20,
          paddingBottom: 8,
        }}
      >
        {messages.map((msg, i) => (
          <div
            key={i}
            style={{ animation: "fadeIn 0.3s ease forwards" }}
          >
            {msg.typing ? (
              <div style={{ display: "flex", padding: "0 16px", marginBottom: 16 }}>
                <div
                  style={{
                    width: 32,
                    height: 32,
                    borderRadius: "50%",
                    background: "linear-gradient(135deg, #c8a96e, #8b6914)",
                    display: "flex",
                    alignItems: "center",
                    justifyContent: "center",
                    fontSize: 13,
                    fontWeight: 700,
                    color: "#0a0a0f",
                    marginRight: 10,
                    flexShrink: 0,
                    marginTop: 2,
                    fontFamily: "'Space Mono', monospace",
                  }}
                >
                  A
                </div>
                <div
                  style={{
                    background: "#13131a",
                    border: "1px solid #1e1e2a",
                    borderRadius: "4px 18px 18px 18px",
                    minWidth: 60,
                  }}
                >
                  <TypingIndicator />
                </div>
              </div>
            ) : (
              <Message msg={msg} />
            )}
          </div>
        ))}
        <div ref={messagesEndRef} />
      </div>

      {/* Quick prompts */}
      {messages.length <= 1 && (
        <div
          style={{
            padding: "8px 0 8px 16px",
            display: "flex",
            gap: 8,
            overflowX: "auto",
            scrollbarWidth: "none",
          }}
        >
          {quickPrompts.map((p) => (
            <QuickPrompt key={p} text={p} onClick={sendMessage} />
          ))}
        </div>
      )}

      {/* Input */}
      <div
        style={{
          padding: "12px 16px 20px",
          borderTop: "1px solid #1a1a24",
          background: "#07070e",
        }}
      >
        <div
          style={{
            display: "flex",
            gap: 10,
            alignItems: "flex-end",
            background: "#0d0d14",
            border: "1px solid #1e1e2a",
            borderRadius: 16,
            padding: "10px 14px",
            transition: "border-color 0.2s",
          }}
          onFocus={(e) => e.currentTarget.style.borderColor = "#c8a96e44"}
          onBlur={(e) => e.currentTarget.style.borderColor = "#1e1e2a"}
        >
          <textarea
            ref={inputRef}
            value={input}
            onChange={(e) => setInput(e.target.value)}
            onKeyDown={handleKeyDown}
            placeholder="Ask about markets, budget, investments..."
            rows={1}
            style={{
              flex: 1,
              background: "transparent",
              border: "none",
              outline: "none",
              color: "#e8e4d8",
              fontSize: 14,
              fontFamily: "'Inter', sans-serif",
              lineHeight: 1.5,
              maxHeight: 120,
              overflowY: "auto",
            }}
            onInput={(e) => {
              e.target.style.height = "auto";
              e.target.style.height = Math.min(e.target.scrollHeight, 120) + "px";
            }}
          />
          <button
            onClick={() => sendMessage()}
            disabled={loading || !input.trim()}
            style={{
              width: 34,
              height: 34,
              borderRadius: "50%",
              background:
                loading || !input.trim()
                  ? "#1a1a24"
                  : "linear-gradient(135deg, #c8a96e, #8b6914)",
              border: "none",
              cursor: loading || !input.trim() ? "not-allowed" : "pointer",
              display: "flex",
              alignItems: "center",
              justifyContent: "center",
              flexShrink: 0,
              transition: "all 0.2s",
              boxShadow:
                loading || !input.trim() ? "none" : "0 0 12px #c8a96e40",
            }}
          >
            <svg
              width="16"
              height="16"
              viewBox="0 0 24 24"
              fill="none"
              stroke={loading || !input.trim() ? "#333344" : "#0a0a0f"}
              strokeWidth="2.5"
              strokeLinecap="round"
              strokeLinejoin="round"
            >
              <line x1="22" y1="2" x2="11" y2="13" />
              <polygon points="22 2 15 22 11 13 2 9 22 2" />
            </svg>
          </button>
        </div>
        <div
          style={{
            textAlign: "center",
            fontSize: 10,
            color: "#222233",
            marginTop: 8,
            fontFamily: "'Space Mono', monospace",
            letterSpacing: "0.05em",
          }}
        >
          NOT FINANCIAL ADVICE · FOR EDUCATIONAL USE
        </div>
      </div>
    </div>
  );
}
